Golf Game PWA v3 — JSONP 跨域补丁

为什么要加这个补丁：
GitHub Pages 与 Google Apps Script 属于不同网域，浏览器直接 fetch() 会遇到 CORS / NetworkError。v3 PWA 改用 JSONP，因此 Apps Script 的 doGet 必须支持 callback 参数。

请在你现在正常工作的 doGet(e) 上再增加下面逻辑。最简单的方式：把当前 doGet(e) 整个替换为以下版本。

function doGet(e) {
  try {
    const p = (e && e.parameter) ? e.parameter : {};
    const action = p.action || '';
    const callback = p.callback || '';

    let result;

    if (action === 'getCourses') {
      result = getCourses_();
    } else if (action === 'getCourse') {
      result = getCourse_(p.course_id);
    } else if (action === 'getMatch') {
      result = getMatch_(p.match_id);
    } else if (action === 'getScores') {
      result = getScores_(p.match_id, p.hole);

    // v3：写操作也通过 GET + JSONP，避免 GitHub Pages CORS。
    } else if (action === 'createCourse') {
      result = createCourse_({
        course_name: p.course_name,
        holes: JSON.parse(p.holes || '[]')
      });
    } else if (action === 'createMatch') {
      result = createMatch_({
        admin: p.admin,
        player_count: Number(p.player_count),
        course_id: p.course_id
      });
    } else if (action === 'joinMatch') {
      result = joinMatch_({
        match_id: p.match_id,
        name: p.name
      });
    } else if (action === 'submitScore') {
      result = submitScore_({
        match_id: p.match_id,
        player_id: p.player_id,
        hole: Number(p.hole),
        score: Number(p.score),
        up: String(p.up) === 'true',
        submitted_by: p.submitted_by || p.player_id
      });
    } else {
      result = {
        success: true,
        message: 'Golf Game API is running'
      };
    }

    if (callback) return jsonp_(callback, result);
    return json_(result);
  } catch (err) {
    const result = {success:false,error:String(err && err.message ? err.message : err)};
    const callback = e && e.parameter ? e.parameter.callback : '';
    if (callback) return jsonp_(callback, result);
    return json_(result);
  }
}

function jsonp_(callback, obj) {
  // callback 只允许安全的 JS 标识符，避免把任意代码当 callback 执行。
  if (!/^[A-Za-z_$][0-9A-Za-z_$\.]*$/.test(callback)) {
    return json_({success:false,error:'Invalid callback'});
  }
  return ContentService
    .createTextOutput(callback + '(' + JSON.stringify(obj) + ');')
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}

重要：
1. 你原来的 doPost(e) 可以保留，不需要删除。
2. 你现在已经能正常工作的 getCourses_ / getCourse_ / createCourse_ 保留。
3. createMatch_、joinMatch_、submitScore_、getMatch_、getScores_ 必须是你现有 Apps Script 中实际存在的函数名。如果你的函数名称不同，只改上面对应的函数调用名称，不要改 PWA。
4. 重新保存后，重新部署 Web App 为“新版本”。正式 /exec URL 可以保持不变。
5. 不需要重新建立 COURSES；现有 COURSES 数据不会删除。

安全提醒：v3 为了解决 GitHub Pages 跨域，把少量写操作改成 URL 参数传给 Apps Script。这个系统只有最多约 8 人使用，因此可以先用于测试；正式投入长期使用前，再加管理员密码/加入码校验和更严格的权限控制。
